import base64,hashlib,json,pathlib,zlib
root=pathlib.Path.cwd()
m=json.loads((root/'release-v62/verified.json').read_text())
parts=[]
for name in m['chunks']:
    text=''.join((root/name).read_text().split())
    repair=m['repairs'].get(name)
    if repair:
        assert hashlib.sha256(text.encode()).hexdigest()==repair['input_sha256'],name
        for a,b,s in reversed(repair['edits']):text=text[:a]+s+text[b:]
    parts.append(text)
blob=base64.b64decode(''.join(parts))
assert hashlib.sha256(blob).hexdigest()==m['payload_sha256'],'Transport checksum mismatch'
sources=json.loads(zlib.decompress(blob))
for name,content in sources.items():
    target=root/name
    assert target.parent==root and not name.startswith('.')
    target.write_text(content,encoding='utf-8')
p=root/'finalize_v62.py'
extra="SOURCE += [str(p.relative_to(ROOT)) for p in sorted((ROOT/'release-v62').glob('payload-*.b64'))] + ['release-v62/install.txt','release-v62/verified.json']\n"
p.write_text(p.read_text().replace('LABELS=',extra+'LABELS=',1))
p=root/'README.md';p.write_text(p.read_text().replace('scan.yml','v62.yml'))
installed={n:(root/n).read_text() for n in sources}
fingerprint=hashlib.sha256(json.dumps(installed,ensure_ascii=False,sort_keys=True,separators=(',',':')).encode()).hexdigest()
assert fingerprint==m['installed_sha256'],'Installed source checksum mismatch'
print('All 24 reviewed source files installed and verified:',fingerprint,flush=True)
